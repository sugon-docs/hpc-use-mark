#!/bin/bash

# 定义函数，递归遍历文件夹
copy_files_recursive() {
    local source_dir="$1"
    local dest_dir="$2"

    # 创建目标文件夹（如果不存在）
    mkdir -p "$dest_dir"

    # 遍历当前文件夹中的文件和文件夹
    for item in "$source_dir"/*; do
        if [ -d "$item" ]; then
            copy_files_recursive "$item" "$dest_dir/$(basename "$item")"
        elif [ -f "$item" ]; then
            filename=$(basename "$item")
        if [[ "$filename" == *.txt || "$filename" == *.out ]]; then  # 拣选出 txt 和 out 后缀
                relative_path=$(dirname "${item#$source_dir}")
                mkdir -p "$dest_dir/$relative_path"
                cp "$item" "$dest_dir/$relative_path/"
            fi
        fi
    done
}

# 开始从当前目录递归遍历，复制符合条件的文件到 ~/data 目录下
copy_files_recursive . ~/data 

echo "复制完成"
