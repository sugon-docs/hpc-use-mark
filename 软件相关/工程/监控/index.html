      #!/bin/bash
#
# CPU利用率监控守护进程
# 功能：当CPU使用率连续10分钟低于5%时，自动取消指定的Slurm作业
# 用法：./cpu_monitor_daemon.sh <jobid> [interval] [threshold]
#
# 参数说明：
#   jobid    : 要监控的Slurm作业ID（必需）
#   interval : 检查间隔时间（秒，默认60秒）
#   threshold: CPU使用率阈值（百分比，默认5%）
#

# 默认参数设置
JOB_ID="$1"
CHECK_INTERVAL="${2:-60}"           # 默认60秒检查一次
CPU_THRESHOLD="${4:-5}"             # 默认5%阈值
MONITOR_DURATION=600                 # 监控时长10分钟（600秒）

# 验证参数
if [ $# -lt 1 ]; then
    echo "错误: 必须提供作业ID参数"
    echo "用法: $0 <jobid> [interval] [threshold]"
    echo "示例: $0 123456 60 5"
    exit 1
fi

# 输入验证
if ! [[ "$JOB_ID" =~ ^[0-9]+$ ]]; then
    echo "错误: 作业ID必须是数字"
    exit 1
fi

if ! [[ "$CHECK_INTERVAL" =~ ^[0-9]+$ ]] || [ "$CHECK_INTERVAL" -lt 10 ]; then
    echo "错误: 检查间隔必须大于等于10秒"
    exit 1
fi

if ! [[ "$CPU_THRESHOLD" =~ ^[0-9]+$ ]] || [ "$CPU_THRESHOLD" -gt 100 ]; then
    echo "错误: CPU阈值必须是0-100之间的整数"
    exit 1
fi

# 计算需要连续低于阈值的检查次数
CHECK_COUNT=$((MONITOR_DURATION / CHECK_INTERVAL))

# 日志功能
LOG_FILE="/tmp/cpu_monitor_${JOB_ID}.log"
log_message() {
    local timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo "[$timestamp] $1" >> "$LOG_FILE"
    echo "[$timestamp] $1"  # 同时输出到控制台
}

# 获取当前CPU使用率（使用top命令）[2](@ref)
get_cpu_usage() {
    local cpu_usage
    cpu_usage=$(top -bn1 | grep "Cpu(s)" | sed "s/.*, *\([0-9.]*\)%* id.*/\1/" | awk '{print 100 - $1}')
    echo "${cpu_usage%%.*}"  # 返回整数部分
}

# 获取当前CPU使用率（备用方法，使用/proc/stat）[1](@ref)
get_cpu_usage_alt() {
    # 第一次采集CPU数据
    local cpu1=$(cat /proc/stat | grep '^cpu ' | awk '{used=$2+$3+$4; unused=$5+$6+$7+$8; print used, unused}')

    sleep 1  # 等待1秒

    # 第二次采集CPU数据
    local cpu2=$(cat /proc/stat | grep '^cpu ' | awk '{used=$2+$3+$4; unused=$5+$6+$7+$8; print used, unused}')

    # 计算CPU使用率
    local usage=$(echo "$cpu1 $cpu2" | awk '
    {
        used1=$1; unused1=$2;
        used2=$3; unused2=$4;
        total_used=used2-used1;
        total_unused=unused2-unused1;
        total=total_used+total_unused;
        if(total > 0) {
            printf("%d\n", (total_used*100)/total);
        } else {
            printf("0\n");
        }
    }')

    echo "$usage"
}

# 检查作业是否仍在运行
is_job_running() {
    squeue -j "$JOB_ID" &>/dev/null
    return $?
}

# 发送取消作业命令
cancel_job() {
    log_message "执行: scancel $JOB_ID"
    scancel "$JOB_ID"
    local result=$?
    if [ $result -eq 0 ]; then
        log_message "成功: 作业 $JOB_ID 已被取消"
    else
        log_message "错误: 取消作业 $JOB_ID 失败 (退出码: $result)"
    fi
    return $result
}

# 初始化
low_cpu_counter=0
check_sequence=0

log_message "启动CPU监控守护进程"
log_message "目标作业: $JOB_ID"
log_message "检查间隔: ${CHECK_INTERVAL}秒"
log_message "CPU阈值: ${CPU_THRESHOLD}%"
log_message "监控时长: ${MONITOR_DURATION}秒"
log_message "需要连续低于阈值的次数: ${CHECK_COUNT}"
log_message "日志文件: ${LOG_FILE}"

# 主监控循环
while true; do
    check_sequence=$((check_sequence + 1))
    current_time=$(date '+%Y-%m-%d %H:%M:%S')

    # 检查作业是否仍在运行
    if ! is_job_running; then
        log_message "作业 $JOB_ID 已结束，监控进程退出"
        exit 0
    fi

    # 获取CPU使用率
    cpu_usage=$(get_cpu_usage)

    # 如果第一种方法失败，使用备用方法
    if [ -z "$cpu_usage" ] || [ "$cpu_usage" -gt 100 ]; then
        cpu_usage=$(get_cpu_usage_alt)
    fi

    # 记录当前状态
    log_message "检查 #${check_sequence}: CPU使用率 = ${cpu_usage}%"

    # 检查CPU使用率是否低于阈值
    if [ "$cpu_usage" -lt "$CPU_THRESHOLD" ]; then
        low_cpu_counter=$((low_cpu_counter + 1))
        log_message "CPU使用率低于阈值 ${CPU_THRESHOLD}% (连续 ${low_cpu_counter}/${CHECK_COUNT} 次)"

        # 检查是否达到连续次数要求
        if [ "$low_cpu_counter" -ge "$CHECK_COUNT" ]; then
            log_message "条件满足: CPU使用率连续${CHECK_COUNT}次检查低于${CPU_THRESHOLD}%"
            log_message "准备取消作业 $JOB_ID ..."

            # 取消作业
            cancel_job
            exit $?
        fi
    else
        if [ "$low_cpu_counter" -gt 0 ]; then
            log_message "CPU使用率高于阈值，重置计数器 (原值: ${low_cpu_counter})"
            low_cpu_counter=0
        fi
    fi

    # 等待下一次检查
    sleep "$CHECK_INTERVAL"
done
