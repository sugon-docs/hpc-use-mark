./configure --prefix=/public/home/eitech002/ThirdCompile/jsy-test/hdf5-1.14.6/install  --enable-fortran --enable-cxx --enable-parallel --enable-shared --enable-unsupported CC=mpicc  CXX=mpicxx  FC=mpifort  CFLAGS="-O3 -march=native"   CXXFLAGS="-O3 -march=native"

指定编译环境

module purge
module load compiler/gcc/9.3.0 mpi/openmpi/openmpi-4.1.5-gcc9.3.0
module list
export CC=mpicc    # C编译器
export CXX=mpicxx  # C++编译器
export FC=mpifort  # Fortran编译器
export F77=mpifort # Fortran77编译器
export F90=mpifort # Fortran90编译器
export CFLAGS="-O3 -march=native"   # 编译选项
export CXXFLAGS="-O3 -march=native" # 编译选项
export HDF5_DIR=/work/home/dabao/soft/hdf5-1.12/install
export PATH=${HDF5_DIR}/bin:$PATH
export LD_LIBRARY_PATH=${HDF5_DIR}/lib:$LD_LIBRARY_PATH
export CPATH=${HDF5_DIR}/include:$CPATH
export CPATH=/public/software/mpi/openmpi/4.1.5/gcc-9.3.0/source/openmpi-4.1.5/ompi/mca/io/romio321/romio/include:$CPATH

env PERL5LIB="" PERL_LOCAL_LIB_ROOT="" PERL_MM_OPT="" PERL_MB_OPT="" cpanm -f Scalar::Util # 清楚别的perl环境
