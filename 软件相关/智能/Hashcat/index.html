安装

# dcu不太清楚
# gpu cpu基本一致
# 安装
git clone https://github.com/hashcat/hashcat.git
# git clone https://kgithub.com/hashcat/hashcat.git # 国内镜像克隆
cd hashcat
make
make install

# 测试
hashcat -I  # 查看硬件信息
hashcat -b -m 22000  # 测试是否正常运行以及破解速度
